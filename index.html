    if after.channel and not before.channel:
        uid = str(member.id)
        HAFIZA_VERI["xp"][uid] = HAFIZA_VERI["xp"].get(uid, 0) + 10
        xp = HAFIZA_VERI["xp"][uid]

            if rol and rol not in member.roles: await member.add_roles(rol)
        elif xp >= 300:
            rol = discord.utils.get(member.guild.roles, name="HÜKÜM Erkanı")
            if rol and rol not in member.roles: await member.add_roles(rol)

class KayitModal(discord.ui.Modal, title='HÜKÜM Giriş Paneli'):
    amac = discord.ui.TextInput(label='Sunucuya ne için geldiniz?', placeholder='Örn: Alışveriş veya Family ortamı...', style=discord.TextStyle.long)

    async def on_submit(self, interaction: discord.Interaction):
        guild = interaction.guild
        member = interaction.user
        
        dogrulandi_rol = discord.utils.get(guild.roles, name="Doğrulandı")
        dogrulanmadi_rol = discord.utils.get(guild.roles, name="Doğrulanmadı")

            await member.add_roles(dogrulandi_rol)
            
        await interaction.response.send_message("✅ Kimliğin doğrulandı aslanım! Şimdi aşağıdan dünyanı seç:", view=DunyaSecimView(), ephemeral=True)

class SatunAlButon(discord.ui.View):
    def __init__(self, ilan_id: str):
        super().__init__(timeout=None)
        self.ilan_id = ilan_id

    @discord.ui.button(label="Satın Al 🛒", style=discord.ButtonStyle.danger, custom_id="gercek_satin_al_butonu")
    async def satin_al_bas(self, interaction: discord.Interaction, button: discord.ui.Button):
        ilan = HAFIZA_VERI["ilanlar"].get(self.ilan_id)
        if not ilan:
            await interaction.response.send_message("❌ Bu ilan tükenmiş veya kaldırılmış!", ephemeral=True)
            return
            
        alici_id = str(interaction.user.id)
        satici_id = ilan["satici_id"]
        fiyat = ilan["fiyat"]
        
        if alici_id == satici_id:
            await interaction.response.send_message("❌ Kendi ürününü satın alamazsın lo!", ephemeral=True)
            return
            
        alici_bakiye = HAFIZA_VERI["jetonlar"].get(alici_id, 0)
        if alici_bakiye < fiyat:
            await interaction.response.send_message(f"❌ Yetersiz Jeton! Gereken: {fiyat}, Sende Olan: {alici_bakiye}", ephemeral=True)
            return
            
        HAFIZA_VERI["jetonlar"][alici_id] = alici_bakiye - fiyat
        HAFIZA_VERI["jetonlar"][satici_id] = HAFIZA_VERI["jetonlar"].get(satici_id, 0) + fiyat
        
        ilan["stok"] -= 1
        if ilan["stok"] <= 0:
            try: await interaction.message.delete()
            except: pass
            del HAFIZA_VERI["ilanlar"][self.ilan_id]
        else:
            HAFIZA_VERI["ilanlar"][self.ilan_id] = ilan
            embed = interaction.message.embeds[0]
            embed.set_field_at(1, name="📦 Stok", value=f"{ilan['stok']} Adet", inline=True)
            await interaction.message.edit(embed=embed)
            
        await interaction.response.send_message(f"🎉 Başarıyla satın aldınız! {fiyat} Jeton satıcıya gönderildi.", ephemeral=True)
        satici = interaction.guild.get_member(int(satici_id))
        if satici:
            try: await satici.send(f"🪙 {interaction.user.name} adlı kişi '{ilan['urun_adi']}' ürününüzü satın aldı! +{fiyat} Jeton hesabınızda.")
            except: pass

            return

class YarismaOySecim(discord.ui.Select):
    def __init__(self, mesaj_id: str, sunucular: list):
        self.mesaj_id = mesaj_id
        options = []
        for i, s in enumerate(sunucular):
            options.append(discord.SelectOption(label=s["ad"], description=f"Oy vermek için tıkla!", emoji="⭐", value=str(i)))
        if not options:
            options.append(discord.SelectOption(label="Henüz Sunucu Yok", value="bos"))
        super().__init__(placeholder="⭐ Oy vermek istediğin sunucuyu seç...", min_values=1, max_values=1, options=options)

    async def callback(self, interaction: discord.Interaction):
        if self.values[0] == "bos":
            await interaction.response.send_message("❌ Önce bir sunucu katılsın paşam!", ephemeral=True)
            return
        oy_kullananlar = HAFIZA_VERI["yarismalar"][self.mesaj_id]["oy_kullananlar"]
        if interaction.user.id in oy_kullananlar:
            await interaction.response.send_message("❌ Sadece 1 kez oy kullanabilirsin aşko!", ephemeral=True)
            return
        secilen_index = int(self.values[0])
        HAFIZA_VERI["yarismalar"][self.mesaj_id]["sunucular"][secilen_index]["oylar"] += 1
        oy_kullananlar.append(interaction.user.id)
        await interaction.response.send_message("✅ Oyun kaydedildi aslanım!", ephemeral=True)

class YarismaKatilButon(discord.ui.Button):
    def __init__(self, mesaj_id: str):
        super().__init__(label="Kendi Sunucumu Yarışa Ekle 🏰", style=discord.ButtonStyle.success, custom_id="yarisma_ekle_butonu_hukum")
        self.mesaj_id = mesaj_id

    async def callback(self, interaction: discord.Interaction):
        sunucular = HAFIZA_VERI["yarismalar"][self.mesaj_id]["sunucular"]
        for s in sunucular:
            if s["ekleyen_id"] == interaction.user.id:
                await interaction.response.send_message("❌ Zaten sunucu eklemişsin lo!", ephemeral=True)
                return
        yeni_sunucu = {"ad": f"{interaction.user.name}'ın Krallığı", "ekleyen_id": interaction.user.id, "oylar": 0}
        sunucular.append(yeni_sunucu)
        
        embed = interaction.message.embeds[0]
        liste_yazi = "".join([f"• **{s['ad']}** — Oy: `{s['oylar']}`\n" for s in sunucular])
        embed.set_field_at(0, name="📊 Katılan Sunucular ve Oylar", value=liste_yazi, inline=False)
        
        yeni_view = discord.ui.View(timeout=None)
        yeni_view.add_item(YarismaOySecim(self.mesaj_id, sunucular))
        yeni_view.add_item(self)
        await interaction.message.edit(embed=embed, view=yeni_view)
        await interaction.response.send_message("🏰 Sunucun eklendi aslanım!", ephemeral=True)
