                    "hapisler": {}, "gardiyan_terfi": {}
                }
                for k, v in varsayilanlar.items():
                    if k not in data:
                        data[k] = v
                return data
            except:
                pass

def get_max_kademe(user: discord.Member) -> int:
    if user.guild_permissions.administrator:
        return 1000
    max_k = 0
    for r in user.roles:
        if r.name in SES_ROLLER_AYAR:
            max_k = max(max_k, SES_ROLLER_AYAR[r.name]["kademe"])
        if r.name in MOD_HİYERARŞİ:
            max_k = max(max_k, MOD_HİYERARŞİ[r.name])
    return max_k

@bot.event
async def on_member_join(member):
    guild = member.guild
    rol = discord.utils.get(guild.roles, name="⚪ Doğrulanmadı")
    if rol:
        try: await member.add_roles(rol)
        except: pass

    eski_davetler = bot.davetler.get(guild.id, [])
    try:
        yeni_davetler = await guild.invites()
        bot.davetler[guild.id] = yeni_davetler
        
        for invite in eski_davetler:
            for yeni_invite in yeni_davetler:
                if invite.code == yeni_invite.code and yeni_invite.uses > invite.uses:
                    inviter = yeni_invite.inviter
                    if inviter:
                        uid_inviter = str(inviter.id)
                        user_invites = sum(i.uses for i in yeni_davetler if i.inviter == inviter)
                        
                        if str(user_invites) in HAFIZA_VERI["davet_odulleri"]:
                            odul = HAFIZA_VERI["davet_odulleri"][str(user_invites)]
                            HAFIZA_VERI["jetonlar"][uid_inviter] = HAFIZA_VERI["jetonlar"].get(uid_inviter, 0) + odul
                            veritabanı_kaydet()
                            
                            kanal = discord.utils.get(guild.text_channels, name="👑-family-rol-kasma")
                            if kanal:
                                await kanal.send(f"🦅 {inviter.mention} nefer getirmeye devam ediyor! `{user_invites}.` Davet ödülü olan **{odul} Jeton** hesaba aktarıldı.")
                    return
    except Exception as e:
        print(f"Davet kontrol hatası: {e}")
