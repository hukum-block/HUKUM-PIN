class RolOnayView(discord.ui.View):
    def __init__(self, rol_isim: str = "", talep_eden_id: int = None):
        super().__init__(timeout=None)
        self.rol_isim = rol_isim
        self.talep_eden_id = talep_eden_id

    @discord.ui.button(label="Evet ✅", style=discord.ButtonStyle.success, custom_id="rol_onay_evet")
    async def onay(self, interaction: discord.Interaction, button: discord.ui.Button):
        if not interaction.user.guild_permissions.administrator:
            await interaction.response.send_message("❌ Bu onayı sadece HÜKÜM Kurmayları verebilir!", ephemeral=True)
            return

        embed = interaction.message.embeds[0] if interaction.message.embeds else None
        rol_adi = self.rol_isim or (embed.title.split(" - ")[1] if embed else None)
        
        if not rol_adi:
            await interaction.response.send_message("❌ Rol bilgisi alınamadı!", ephemeral=True)
            return
            
        guild = interaction.guild
        
        target_member = None
        if self.talep_eden_id:
            target_member = guild.get_member(self.talep_eden_id)
        elif interaction.message.mentions:
            target_member = interaction.message.mentions[0]

        if not target_member:
            await interaction.response.send_message("❌ Rütbe sahibi sunucuda bulunamadı!", ephemeral=True)
            return

        rol = discord.utils.get(guild.roles, name=rol_adi)
        
        if rol:
            if rol in target_member.roles:
                await interaction.response.send_message("❌ Bu rol zaten bu yiğitte var paşam!", ephemeral=True)
            else:
                await target_member.add_roles(rol)
                await interaction.response.send_message(f"🎉 {target_member.mention} rütbesi teslim edildi: **{rol_adi}**! Dik dur aslanım!", ephemeral=False)
                try:
                    await target_member.send(f"👑 **MÜJDE PAŞAM!** {interaction.guild.name} sunucusunda talep ettiğin **{rol_adi}** rütben yönetim tarafından onaylandı! Sancağı dik!")
                except:
                    pass
                await interaction.message.delete()
        else:
            await interaction.response.send_message("❌ Sunucuda rol bulunamadı!", ephemeral=True)

    @discord.ui.button(label="Hayır ❌", style=discord.ButtonStyle.danger, custom_id="rol_onay_hayir")

@bot.event
async def on_voice_state_update(member, before, after):
    if member.bot:
        return
    uid = str(member.id)
    simdi = datetime.datetime.now().timestamp()

    if after.channel and not before.channel:
        HAFIZA_VERI["ses_giris"][uid] = simdi
        veritabanı_kaydet()

    elif before.channel and not after.channel:
        giris_zamani = HAFIZA_VERI["ses_giris"].pop(uid, simdi)
        gecen_sure = simdi - giris_zamani
        HAFIZA_VERI["ses_sureleri"][uid] = HAFIZA_VERI["ses_sureleri"].get(uid, 0) + gecen_sure
        veritabanı_kaydet()

        total_saat = HAFIZA_VERI["ses_sureleri"][uid] / 3600
        
        for r_isim, sartlar in SES_ROLLER_AYAR.items():
            if total_saat >= sartlar["ses"]:
                rol = discord.utils.get(member.guild.roles, name=r_isim)
                if rol and rol not in member.roles:
                    kanal = discord.utils.get(member.guild.text_channels, name="👑-family-rol-kasma")
                    if kanal:
                        embed = discord.Embed(
                            title=f"👑 RÜTBE KAZANIMI - {r_isim}", 
                            description=f"Hey {member.mention}! Seste kaldığın süre (`{total_saat:.1f}` saat) ile bu rolü almaya hak kazandın. İster misin?", 
                            color=discord.Color.green()
                        )
                        await kanal.send(content=member.mention, embed=embed, view=RolOnayView(r_isim, member.id))
                break

class SatunAlButon(discord.ui.View):
    def __init__(self, ilan_id: str):
        super().__init__(timeout=None)
        self.ilan_id = ilan_id

    @discord.ui.button(label="Satın Al 🛒", style=discord.ButtonStyle.danger, custom_id="gercek_satin_al_butonu")
    async def satin_al_bas(self, interaction: discord.Interaction, button: discord.ui.Button):
        ilan = HAFIZA_VERI["ilanlar"].get(self.ilan_id)
        if not ilan:
            await interaction.response.send_message("❌ Bu ilan tükenmiş veya kaldırılmış!", ephemeral=True)
            return
        alici_id = str(interaction.user.id)
        satici_id = ilan["satici_id"]
        fiyat = ilan["fiyat"]
        if alici_id == satici_id:
            await interaction.response.send_message("❌ Kendi ürününü satın alamazsın lo!", ephemeral=True)
            return
        alici_bakiye = HAFIZA_VERI["jetonlar"].get(alici_id, 0)
        if alici_bakiye < fiyat:
            await interaction.response.send_message(f"❌ Yetersiz Jeton! Gereken: {fiyat}, Sende Olan: {alici_bakiye}", ephemeral=True)
            return
        HAFIZA_VERI["jetonlar"][alici_id] = alici_bakiye - fiyat
        HAFIZA_VERI["jetonlar"][satici_id] = HAFIZA_VERI["jetonlar"].get(satici_id, 0) + fiyat
        ilan["stok"] -= 1
        if ilan["stok"] <= 0:
            try:
                await interaction.message.delete()
            except:
                pass
            del HAFIZA_VERI["ilanlar"][self.ilan_id]
        else:
            HAFIZA_VERI["ilanlar"][self.ilan_id] = ilan
            embed = interaction.message.embeds[0]
            embed.set_field_at(1, name="📦 Stok", value=f"{ilan['stok']} Adet", inline=True)
            await interaction.message.edit(embed=embed)
        veritabanı_kaydet()
        await interaction.response.send_message(f"🎉 Başarıyla satın aldınız! {fiyat} Jeton satıcıya gönderildi.", ephemeral=True)
        satici = interaction.guild.get_member(int(satici_id))
        if satici:
            try:
                await satici.send(f"🪙 {interaction.user.name} adlı kişi '{ilan['urun_adi']}' ürününüzü satın aldı! +{fiyat} Jeton hesabınızda.")
            except:
                pass

class YarismaOySecim(discord.ui.Select):
    def __init__(self, mesaj_id: str, sunucular: list):
        self.mesaj_id = mesaj_id
        options = []
        for i, s in enumerate(sunucular):
            options.append(discord.SelectOption(label=s["ad"], description=f"Oy vermek için tıkla!", emoji="⭐", value=str(i)))
        if not options:
            options.append(discord.SelectOption(label="Henüz Sunucu Yok", value="bos"))
        super().__init__(placeholder="⭐ Oy vermek istediğin sunucuyu seç...", min_values=1, max_values=1, options=options)

    async def callback(self, interaction: discord.Interaction):
        if self.values[0] == "bos":
            await interaction.response.send_message("❌ Önce bir sunucu katılsın paşam!", ephemeral=True)
            return
        if self.mesaj_id not in HAFIZA_VERI["yarismalar"]:
            await interaction.response.send_message("❌ Bu yarışma sonlandı lo!", ephemeral=True)
            return
            
        oy_kullananlar = HAFIZA_VERI["yarismalar"][self.mesaj_id]["oy_kullananlar"]
        if interaction.user.id in oy_kullananlar:
            await interaction.response.send_message("❌ Sadece 1 kez oy kullanabilirsin!", ephemeral=True)
            return
        secilen_index = int(self.values[0])
        HAFIZA_VERI["yarismalar"][self.mesaj_id]["sunucular"][secilen_index]["oylar"] += 1
        oy_kullananlar.append(interaction.user.id)
        veritabanı_kaydet()
        
        sunucular = HAFIZA_VERI["yarismalar"][self.mesaj_id]["sunucular"]
        embed = interaction.message.embeds[0]
        liste_yazi = "".join([f"• **{s['ad']}** — Oy: `{s['oylar']}` (Ekleyen: <@{s['ekleyen_id']}>)\n" for s in sunucular])
        embed.set_field_at(0, name="📊 Katılan Sunucular ve Oylar", value=liste_yazi, inline=False)
        await interaction.message.edit(embed=embed)
        await interaction.response.send_message("✅ Oyun başarıyla kaydedildi aslanım!", ephemeral=True)

# ================= ARKA PLAN KURULUM ASYNC GÖREVİ (RATE-LIMIT SAVAR KUSURSUZ MOTOR) =================
async def arka_plan_kurulum(guild, interaction):
    # 1. ESKİ KANALLARI VE KATEGORİLERİ TEMİZLEME
    for channel in guild.channels:
        try:
            await channel.delete()
            await asyncio.sleep(0.4) # Silme işlemlerinde de rate limit koruması paşam

            del HAFIZA_VERI["ilanlar"][self.ilan_id]

    kanal = discord.utils.get(interaction.guild.text_channels, name="🎉-çekilişler") or interaction.channel
    await interaction.response.send_message(f"🎉 Çekiliş başladı aslanım!", ephemeral=True)

    cekilis_mesaj = await kanal.send(embed=embed)
    mesaj_id = str(cekilis_mesaj.id)
    
    HAFIZA_VERI["cekilisler"][mesaj_id] = {"odul": odul, "kazanan_sayisi": kazanan_sayisi, "katilanlar": []}
    veritabanı_kaydet()
    
    await cekilis_mesaj.edit(view=CekilisKatilButon(mesaj_id))
    await asyncio.sleep(sure_saniye)
    
    if mesaj_id in HAFIZA_VERI["cekilisler"]:
        katilanlar = HAFIZA_VERI["cekilisler"][mesaj_id]["katilanlar"]
        if len(katilanlar) == 0:
            await cekilis_mesaj.edit(content="❌ Kimse katılmadı!", view=None)
        else:
            kazananlar = random.sample(katilanlar, min(len(katilanlar), kazanan_sayisi))
            mentions = ", ".join([f"<@{k_id}>" for k_id in kazananlar])
            await kanal.send(f"📢 Şampiyonlar: {mentions}!")
        del HAFIZA_VERI["cekilisler"][mesaj_id]
        veritabanı_kaydet()
