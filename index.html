class RolOnayView(discord.ui.View):
    def __init__(self, rol_isim: str = "", talep_eden_id: int = None):
        super().__init__(timeout=None)
        self.rol_isim = rol_isim
        self.talep_eden_id = talep_eden_id

    @discord.ui.button(label="Evet ✅", style=discord.ButtonStyle.success, custom_id="rol_onay_evet_yeni")
    async def onay(self, interaction: discord.Interaction, button: discord.ui.Button):
        if not ozel_yetki_kontrol(interaction):
            await interaction.response.send_message("❌ Bu onayı sadece HÜKÜM Kurmayları verebilir!", ephemeral=True)
            return

        rol_adi = self.rol_isim
        guild = interaction.guild
        target_member = guild.get_member(self.talep_eden_id)

        if not target_member:
            await interaction.response.send_message("❌ Rütbe sahibi sunucuda bulunamadı!", ephemeral=True)
            return

        rol = discord.utils.get(guild.roles, name=rol_adi)
        if rol:
            if rol in target_member.roles:
                await interaction.response.send_message("❌ Bu rol zaten bu yiğitte var paşam!", ephemeral=True)
            else:
                await target_member.add_roles(rol)
                await interaction.response.send_message(f"🎉 {target_member.mention} rütbesi teslim edildi: **{rol_adi}**! Dik dur aslanım!", ephemeral=False)
                try: await target_member.send(f"👑 **MÜJDE PAŞAM!** Talep ettiğin **{rol_adi}** rütben onaylandı!")
                except: pass
                await interaction.message.delete()

    @discord.ui.button(label="Hayır ❌", style=discord.ButtonStyle.danger, custom_id="rol_onay_hayir_yeni")
    async def ret(self, interaction: discord.Interaction, button: discord.ui.Button):

            await interaction.response.send_message("❌ Bu reddi sadece HÜKÜM Kurmayları verebilir!", ephemeral=True)
            return
        await interaction.response.send_message("⚙️ İşlem iptal edildi, rütbe reddedildi.", ephemeral=True)
        await interaction.message.delete()

# OTOMATİK SES TAKİP MOTORU
@bot.event
async def on_voice_state_update(member, before, after):
    if member.bot: return
    uid = str(member.id)
    simdi = datetime.datetime.now().timestamp()

    if after.channel and not before.channel:
        HAFIZA_VERI["ses_giris"][uid] = simdi
        veritabanı_kaydet()
    elif before.channel and not after.channel:
        giris_zamani = HAFIZA_VERI["ses_giris"].pop(uid, simdi)
        gecen_sure = simdi - giris_zamani
        HAFIZA_VERI["ses_sureleri"][uid] = HAFIZA_VERI["ses_sureleri"].get(uid, 0) + gecen_sure
        veritabanı_kaydet()

        total_saat = HAFIZA_VERI["ses_sureleri"][uid] / 3600
        for r_isim, sartlar in SES_ROLLER_AYAR.items():
            if total_saat >= sartlar["ses"]:
                rol = discord.utils.get(member.guild.roles, name=r_isim)
                if rol and rol not in member.roles:
                    kanal = discord.utils.get(member.guild.text_channels, name="👑-family-rol-kasma")
                    if kanal:
                        embed = discord.Embed(
                            title=f"👑 RÜTBE KAZANIMI - {r_isim}", 
                            description=f"Hey {member.mention}! Seste kaldığın süre (`{total_saat:.1f}` saat) ile bu rolü almaya hak kazandın.", 
                            color=discord.Color.green()
                        )
                        await kanal.send(content=member.mention, embed=embed, view=RolOnayView(r_isim, member.id))
                break

class YarismaOySecim(discord.ui.Select):
    def __init__(self, mesaj_id: str, sunucular: list):
        self.mesaj_id = mesaj_id
        options = [discord.SelectOption(label=s["ad"], description="Oy vermek için tıkla!", emoji="⭐", value=str(i)) for i, s in enumerate(sunucular)]
        if not options: options.append(discord.SelectOption(label="Henüz Sunucu Yok", value="bos"))
        super().__init__(placeholder="⭐ Oy vermek istediğin sunucuyu seç...", min_values=1, max_values=1, options=options)

    async def callback(self, interaction: discord.Interaction):
        if self.values[0] == "bos":
            await interaction.response.send_message("❌ Önce bir sunucu katılsın paşam!", ephemeral=True)
            return
        mid = str(interaction.message.id)
        if mid not in HAFIZA_VERI["yarismalar"]:
            await interaction.response.send_message("❌ Bu yarışma sonlandı lo!", ephemeral=True)
            return
            
        oy_kullananlar = HAFIZA_VERI["yarismalar"][mid]["oy_kullananlar"]
        if interaction.user.id in oy_kullananlar:
            await interaction.response.send_message("❌ Sadece 1 kez oy kullanabilirsin!", ephemeral=True)
            return
        secilen_index = int(self.values[0])
        HAFIZA_VERI["yarismalar"][mid]["sunucular"][secilen_index]["oylar"] += 1
        oy_kullananlar.append(interaction.user.id)
        veritabanı_kaydet()
        
        sunucular = HAFIZA_VERI["yarismalar"][mid]["sunucular"]
        embed = interaction.message.embeds[0]
        liste_yazi = "".join([f"• **{s['ad']}** — Oy: `{s['oylar']}` (Ekleyen: <@{s['ekleyen_id']}>)\n" for s in sunucular])
        embed.set_field_at(0, name="📊 Katılan Sunucular ve Oylar", value=liste_yazi, inline=False)
        await interaction.message.edit(embed=embed)
        await interaction.response.send_message("✅ Oyun başarıyla kaydedildi aslanım!", ephemeral=True)

class YarismaKatilButon(discord.ui.Button):
    def __init__(self, mesaj_id: str):
        super().__init__(label="Kendi Sunucumu Yarışa Ekle 🏰", style=discord.ButtonStyle.success, custom_id="yarisma_ekle_butonu_hukum_yeni")
        self.mesaj_id = mesaj_id

    async def callback(self, interaction: discord.Interaction):
        mid = str(interaction.message.id)
        if mid not in HAFIZA_VERI["yarismalar"]:
            await interaction.response.send_message("❌ Yarışma süresi dolmuş lo!", ephemeral=True)
            return
        sunucular = HAFIZA_VERI["yarismalar"][mid]["sunucular"]
        for s in sunucular:
            if s["ekleyen_id"] == interaction.user.id:
                await interaction.response.send_message("❌ Zaten kendi krallığını eklemişsin lo!", ephemeral=True)
                return
        yeni_sunucu = {"ad": f"{interaction.user.name}'ın Krallığı", "ekleyen_id": interaction.user.id, "oylar": 0}
        sunucular.append(yeni_sunucu)
        veritabanı_kaydet()
        
        embed = interaction.message.embeds[0]
        liste_yazi = "".join([f"• **{s['ad']}** — Oy: `{s['oylar']}`\n" for s in sunucular])
        embed.set_field_at(0, name="📊 Katılan Sunucular ve Oylar", value=liste_yazi, inline=False)
        
        yeni_view = discord.ui.View(timeout=None)
        yeni_view.add_item(YarismaOySecim(mid, sunucular))
        yeni_view.add_item(self)
        await interaction.message.edit(embed=embed, view=yeni_view)
        await interaction.response.send_message("🏰 Sunucun başarıyla yarış meydanına eklendi aslanım!", ephemeral=True)
